Agar tumhe sirf storage/app ke root me directly jo .xlsx files hain unko tracking se hatana hai, storage/app/backEnd/... ko nahi, to ye safer way use karo.

Pehle exact root-level tracked files dekho:

git ls-files storage/app | findstr /R "^storage/app/[^/]*\.xlsx$"

public function timesheetnotfilledlastweek(Request $request)
{
    // 1. Previous week ka range
    $previousWeekStart = now()->subWeeks(1)->startOfWeek(); // Monday
    $previousWeekEnd = now()->subWeeks(1)->endOfWeek();     // Sunday

    // 2. Latest team role history (staff code ke liye)
    $latestHistory = DB::table('teamrolehistory as teamrolehistorydata')
        ->select('teamrolehistorydata.teammember_id', 'teamrolehistorydata.newstaff_code')
        ->whereRaw('teamrolehistorydata.id = (SELECT MAX(teamrolehistorylatest.id) FROM teamrolehistory as teamrolehistorylatest WHERE teamrolehistorylatest.teammember_id = teamrolehistorydata.teammember_id)');

    // 3. Sabhi active members fetch karo
    $allMembers = DB::table('teammembers')
        ->leftJoinSub($latestHistory, 'latestrecord', function ($join) {
            $join->on('latestrecord.teammember_id', '=', 'teammembers.id');
        })
        ->where('teammembers.status', 1)
        ->select(
            'teammembers.id',
            'teammembers.emailid',
            'teammembers.team_member',
            'teammembers.staffcode',
            'latestrecord.newstaff_code'
        )
        ->get();

    // 4. Har member ke liye check karo
    foreach ($allMembers as $user) {
        // ---------- Rejoining Date Fetch ----------
        $rejoiningDate = $this->getRejoiningDate($user->id);

        // ---------- Previous Week Timesheet Check (Rejoin ke baad wali) ----------
        $hasSubmitted = DB::table('timesheetusers')
            ->where('createdby', $user->id)
            ->whereBetween('date', [$previousWeekStart, $previousWeekEnd])
            ->where('status', '!=', 0);

        // 🔥 IMPORTANT: Agar rejoin hua hai toh rejoin date ke baad ki entries check karo
        if ($rejoiningDate) {
            $hasSubmitted->where('date', '>=', $rejoiningDate);
        }

        $hasSubmitted = $hasSubmitted->exists();

        if (!$hasSubmitted) {
            // Last overall submission date fetch (rejoin ke baad wali)
            $lastSubmission = DB::table('timesheetusers')
                ->where('createdby', $user->id)
                ->where('status', '!=', 0);

            // 🔥 Sirf rejoin ke baad ki submissions count karo
            if ($rejoiningDate) {
                $lastSubmission->where('date', '>=', $rejoiningDate);
            }

            $lastSubmission = $lastSubmission->max('date');

            $user->last_submission_date = $lastSubmission
                ? Carbon::parse($lastSubmission)->format('d-m-Y')
                : 'Never';
            $user->has_submitted = false;
            $user->rejoining_date = $rejoiningDate;
        } else {
            $user->last_submission_date = '';
            $user->has_submitted = true;
            $user->rejoining_date = $rejoiningDate;
        }
    }

    // 5. Excel Data Prepare (Sirf un members ko jo submit nahi kiye aur Never nahi hain)
    $excelData = $allMembers->filter(function ($user) {
        return !$user->has_submitted && $user->last_submission_date !== 'Never';
    })->map(function ($user) {
        return [
            'team_member' => $user->team_member,
            'emailid' => $user->emailid,
            'staffcode' => $user->newstaff_code ?? ($user->staffcode ?? ''),
            'last_submission_date' => $user->last_submission_date,
            'rejoining_date' => $user->rejoining_date ?? 'N/A',
        ];
    })->toArray();

    // 6. Excel Export
    if (!empty($excelData)) {
        $export = new TimesheetLastWeekExport(collect($excelData));
        $excelFileName = 'Timesheet_not_filled_last_week.xlsx';
        Excel::store($export, $excelFileName);

        // 7. Email Data
        $emailData = array(
            'subject' => "Timesheet Not Filled Last Week",
            'teammember' => $allMembers->filter(function ($user) {
                return !$user->has_submitted && $user->last_submission_date !== 'Never';
            })->map(function ($user) {
                return (object) [
                    'team_member' => $user->team_member,
                    'emailid' => $user->emailid,
                    'staffcode' => $user->newstaff_code ?? ($user->staffcode ?? ''),
                    'last_submission_date' => $user->last_submission_date,
                    'rejoining_date' => $user->rejoining_date ?? 'N/A',
                ];
            }),
        );

        // 8. Send Email
        Mail::send('emails.timesheetnotfilledlastweekreminder', $emailData, function ($msg) use ($emailData, $excelFileName) {
            $msg->to('shahidraza@capitall.io');
            // $msg->to('itsupport_delhi@vsa.co.in');
            // $msg->cc('Admin_delhi@vsa.co.in');
            $msg->attach(storage_path('app/' . $excelFileName), [
                'as' => $excelFileName,
                'mime' => 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
            ]);
            $msg->subject($emailData['subject']);
        });

        return back()->with('success', 'We have sent the Excel file successfully on mail');
    } else {
        Log::info('No data available for Excel generation.');
        return back()->with('info', 'No members found who haven\'t filled timesheet.');
    }
}

/**
 * Get rejoining date for a member
 * Checks both teamrolehistory and rejoiningsamepost tables
 */
private function getRejoiningDate($memberId)
{
    // 1. Latest rejoining date from teamrolehistory
    $latestTeamRoleHistory = DB::table('teamrolehistory as trh')
        ->select('trh.rejoiningdate')
        ->where('trh.teammember_id', $memberId)
        ->whereRaw('trh.id = (SELECT MAX(id) FROM teamrolehistory WHERE teammember_id = trh.teammember_id)')
        ->first();

    // 2. Latest rejoining date from rejoiningsamepost
    $latestRejoiningSamePost = DB::table('rejoiningsamepost as rsp')
        ->select('rsp.rejoiningdate as samepostrejoiningdate')
        ->where('rsp.teammember_id', $memberId)
        ->whereRaw('rsp.id = (SELECT MAX(id) FROM rejoiningsamepost WHERE teammember_id = rsp.teammember_id)')
        ->first();

    $date1 = $latestTeamRoleHistory->rejoiningdate ?? null;
    $date2 = $latestRejoiningSamePost->samepostrejoiningdate ?? null;

    // Return the maximum (latest) date
    if ($date1 && $date2) {
        return max($date1, $date2);
    }

    return $date1 ?? $date2;
}

public function getFileUrl($fileName, $localPath = 'backEnd/image/teammember')
    {
        if (empty($fileName)) {
            return $localPath == 'backEnd/image/teammember/profilepic'
                ? asset('backEnd/image/dummy-user.png')
                : '#';
        }

        // agar temp path aa raha hai to usko ignore karo
        if (
            str_contains($fileName, 'xampp/tmp') ||
            str_contains($fileName, 'xampp\\tmp') ||
            str_contains($fileName, '.tmp')
        ) {
            return '#';
        }

        $fileName = str_replace('\\', '/', trim($fileName));

        $localFile = public_path($localPath . '/' . basename($fileName));

        if (file_exists($localFile)) {
            return asset($localPath . '/' . basename($fileName));
        }

        try {
            if (Storage::disk('s3')->exists($fileName)) {
                return Storage::disk('s3')->temporaryUrl($fileName, now()->addMinutes(30));
            }

            $s3Path = 'candidateonboarding/' . basename($fileName);

            if (Storage::disk('s3')->exists($s3Path)) {
                return Storage::disk('s3')->temporaryUrl($s3Path, now()->addMinutes(30));
            }
        } catch (\Exception $e) {
            \Log::error('MinIO file error: ' . $e->getMessage());
        }

        return $localPath == 'backEnd/image/teammember/profilepic'
            ? asset('backEnd/image/teammember/profilepic/16092382843.png')
            : '#';
    }

        if (empty($fileName)) {
            return '#';
        }

        // agar temp path aa raha hai to usko ignore karo
        if (
            str_contains($fileName, 'xampp/tmp') ||
            str_contains($fileName, 'xampp\\tmp') ||
            str_contains($fileName, '.tmp')
        ) {
            return '#';
        }

        $fileName = str_replace('\\', '/', trim($fileName));

        $localFile = public_path($localPath . '/' . basename($fileName));

        if (file_exists($localFile)) {
            return asset($localPath . '/' . basename($fileName));
        }

        try {
            if (Storage::disk('s3')->exists($fileName)) {
                return Storage::disk('s3')->temporaryUrl($fileName, now()->addMinutes(30));
            }

            $s3Path = 'candidateonboarding/' . basename($fileName);

            if (Storage::disk('s3')->exists($s3Path)) {
                return Storage::disk('s3')->temporaryUrl($s3Path, now()->addMinutes(30));
            }
        } catch (\Exception $e) {
            \Log::error('MinIO file error: ' . $e->getMessage());
        }

        return '#';
    }

 public function allteamsubmitted()
  {
    $get_date = DB::table('timesheetreport as t')
      ->leftJoin('teammembers as tm', 'tm.id', '=', 't.teamid')
      ->leftJoin('teammembers as p', 'p.id', '=', 't.partnerid')
      ->select(
        DB::raw('MIN(t.id) as id'),
        't.teamid',
        't.week',
        DB::raw('SUM(t.totaldays) as totaldays'),
        DB::raw('SUM(t.totaltime) as totaltime'),
        DB::raw('SUM(t.dayscount) as dayscount'),
        DB::raw('MIN(t.startdate) as startdate'),
        DB::raw('MAX(t.enddate) as enddate'),
        DB::raw('MAX(t.created_at) as created_at'),
        'tm.team_member',
        'tm.emailid',
        DB::raw('MAX(p.team_member) as partnername'),
        DB::raw('MAX(t.partnerid) as partnerid')
      )
      ->groupBy('t.teamid', 't.week', 'tm.team_member', 'tm.emailid')
      ->orderByDesc(DB::raw('MAX(t.created_at)'))
      ->limit(50)
      ->get();

    $get_date->each(function ($item) {
      $item->staffcode = $this->getStaffCodeAt($item->teamid, $item->created_at);
    });

    $permissiontimesheet = DB::table('timesheetreport')->first();

    return view('backEnd.timesheet.myteamindex', compact('get_date', 'permissiontimesheet'));
  }

 public function handle($request, Closure $next)
    {
        $user = auth()->user();
        // Check if user is authenticated and inactive
        if (auth()->check() && $user->status == 0) {

            auth()->logout();
            return redirect()->route('login')->withErrors('Your account has been deactivated.');
        }

        if (auth()->check() && $user->two_factor_code && !in_array($user->email, ['Poojakumari@capitall.io', 'manojsangwan@capitall.io', 'sukhbahadur1993@gmail.com', 'sangwanm242@gmail.com', 'Guneet@capitall.io', 'Harish@capitall.io', 'Simmi@capitall.io', 'shahidraza@capitall.io'])) {
            if ($user->two_factor_expires_at->lt(now())) {
                $user->resetTwoFactorCode();
                auth()->logout();

                return redirect()->route('login')
                    ->withMessage('The two factor code has expired. Please login again.');
            }

            if (!$request->is('verify*')) {
                return redirect()->route('verify.index');
            }
        }

        return $next($request);
    }
